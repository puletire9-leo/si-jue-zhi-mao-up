# 阶段四：迭代层 — 时间序列对比与知识演化

> 每周新数据进来，自动对比同产品历史分析，生成变化摘要，追加到知识库。实现从"一次性分析"到"持续追踪"的跨越。

---

## 1. 核心设计：时间序列知识链

```
同一产品 B0DJVXKRB7 的知识链：

Week 20:  [首次分析] 上架 30 天, 月销 2,100, BSR #2,500, 评级 B ──▶ Milvus
Week 21:  [迭代分析] 上架 37 天, 月销 3,200(+52%), BSR #1,200(↓52%), 评级 A ──▶ Milvus
Week 22:  [迭代分析] 上架 44 天, 月销 2,700(-15.6%), BSR #1,800(↑50%), 评级 A ──▶ Milvus
Week 23:  ← 用户问："B0DJVXKRB7 这周跌了吗？"
           → RAG 检索到三份分析，Agent 综合回答
```

**关键点**：不是覆盖旧分析，而是**追加**。同一产品的每一周分析都是独立的知识条目，RAG 检索时可以拿到完整的时间线。

```python
"""跨周产品对比引擎 — 对比新旧数据，生成变化摘要"""

from dataclasses import dataclass, field
from typing import Optional


@dataclass
class MetricChange:
    """单个指标的变化"""
    metric_name: str        # 显示名称："月销量"
    field_name: str         # 字段名："units"
    old_value: float
    new_value: float
    change_pct: float       # 变化百分比: +52.3 或 -15.6
    direction: str          # "up" / "down" / "flat"
    significance: str       # "significant" / "moderate" / "negligible"
    interpretation: str     # 一句话解读


@dataclass
class WeeklyChangeReport:
    """周度变化报告"""
    asin: str
    marketplace: str
    product_title: str
    current_week: str       # "2026-W22"
    previous_week: str      # "2026-W21"
    
    # 变化指标
    changes: list[MetricChange] = field(default_factory=list)
    
    # LLM 生成的变化摘要
    change_summary: str = ""        # 自然语言概述
    key_findings: list[str] = field(default_factory=list)  # 关键发现
    risk_changes: list[str] = field(default_factory=list)  # 风险变化
    
    # 对比上一份分析报告的变化
    analysis_evolution: str = ""    # 分析建议的演变


# ==================== 对比阈值配置 ====================

# 变化显著性阈值
SIGNIFICANT_THRESHOLDS = {
    "units":        {"up": 20, "down": -15},    # 销量: +20% 显著增, -15% 显著降

    "revenue":      {"up": 20, "down": -15},
```

### 2.2 对比计算

```python
# weekly_comparison.py 续

def compute_weekly_changes(
    current_data: dict,
    previous_data: dict,
    product_title: str = "",
    marketplace: str = "UK",
) -> WeeklyChangeReport:
    """
    计算两个周之间的产品指标变化。
    
    Args:
        current_data: 本周的 competitor_products 记录
        previous_data: 上周的 competitor_products 记录
        product_title: 产品标题
        marketplace: 站点
    
    Returns:
        WeeklyChangeReport 包含所有指标的变化
    """
    # 获取当前周标签
    from datetime import datetime
    year = datetime.utcnow().isocalendar()[0]
    week = datetime.utcnow().isocalendar()[1]
    current_week = f"{year}-W{week:02d}"
    
    # 上一周（简化：从 previous_data 的 month 推断）
    prev_month = previous_data.get("month", "unknown")
    previous_week = f"prev-{prev_month}"
    
    report = WeeklyChangeReport(
        asin=current_data.get("asin", ""),
        marketplace=marketplace,
        product_title=product_title,
        current_week=current_week,
        previous_week=previous_week,
    )
    
    # 对比的字段列表
    compare_fields = [
        ("月销量", "units"),
        ("销量增长率", "unitsGr"),
        ("BSR排名", "bsr"),
        ("BSR变化率", "bsrCr"),
        ("售价", "price"),
        ("评价数", "ratings"),
        ("评分", "rating"),
        ("收入", "revenue"),
        ("卖家数", "sellers"),
    ]
    
    for metric_name, field_name in compare_fields:
        old_val = _safe_float(previous_data.get(field_name))
        new_val = _safe_float(current_data.get(field_name))
        
        if old_val == 0 and new_val == 0:
            continue
        
        change = _calculate_change(metric_name, field_name, old_val, new_val)
        if change:
            report.changes.append(change)
    
    return report


def _safe_float(val) -> float:
    """安全转换为 float"""
    if val is None:
        return 0.0
    try:
        return float(val)
    except (ValueError, TypeError):
        return 0.0


def _calculate_change(
    metric_name: str,
    field_name: str,
    old_val: float,
    new_val: float,
) -> Optional[MetricChange]:
    """计算单个指标的变化"""
    
    # 变化百分比
    if old_val != 0:
        change_pct = round((new_val - old_val) / abs(old_val) * 100, 1)
    else:
        change_pct = 100.0 if new_val > 0 else 0.0
    
    # 方向判定
    if abs(change_pct) < 2:
        direction = "flat"
    elif change_pct > 0:
        direction = "up"
    else:
        direction = "down"
    
    # 显著性判定
    thresholds = SIGNIFICANT_THRESHOLDS.get(field_name, {"up": 30, "down": -30})
    if change_pct >= thresholds.get("up", 30):
        significance = "significant"
    elif change_pct <= thresholds.get("down", -30):
        significance = "significant"
    elif abs(change_pct) >= 10:
        significance = "moderate"
    else:
        significance = "negligible"
    
    # 一句话解读
    desc = DIRECTION_DESC.get(field_name, {})
    dir_text = desc.get(direction, f"变化 {change_pct:+.1f}%")
    is_bad = desc.get("up_bad")
    
    if significance == "significant":
        if is_bad is True and direction == "up":
            interpretation = f"⚠️ {dir_text}，幅度较大，需关注"
        elif is_bad is True and direction == "down":
            interpretation = f"✅ {dir_text}，改善明显"
        elif is_bad is False and direction == "up":
            interpretation = f"✅ {dir_text}，表现强劲"
        elif is_bad is False and direction == "down":
            interpretation = f"⚠️ {dir_text}，幅度较大，需关注"
        else:
            interpretation = f"📊 {dir_text}，变化显著"
    elif significance == "moderate":
        interpretation = f"📊 {dir_text}"
    else:
        interpretation = f"➡️ {dir_text}"
    
    return MetricChange(
        metric_name=metric_name,
        field_name=field_name,
        old_value=old_val,
        new_value=new_val,
        change_pct=change_pct,
        direction=direction,
        significance=significance,
        interpretation=interpretation,
    )
```

## 3. LLM 生成变化摘要

```python
# weekly_comparison.py 续

from langchain.chat_models import init_chat_model
import os

CHANGE_SUMMARY_PROMPT = """你是一位跨境电商产品分析专家。以下是同一产品连续两周的数据对比。

## 产品信息
- ASIN: {asin}
- 标题: {title}
- 当前周: {current_week}
- 对比周: {previous_week}

## 指标变化

{changes_text}

## 上一份分析报告的摘要
{previous_summary}

## 任务

请生成一份本周变化摘要，包含：

1. **总体判断**：用一句话概括本周走势（上升 / 平稳 / 下滑）
2. **关键变化**：列出 2-3 个最值得关注的变化（优先关注显著性高 + 方向恶化的指标）
3. **风险更新**：上一份报告提到的风险是否应验？是否有新风险出现？
4. **建议调整**：基于本周数据，之前的建议是否仍然有效？是否需要调整？
5. **关注点**：下周需要重点关注的指标

请用中文输出，风格专业但直接。不要重复完整数据，只讲变化。
"""


def generate_change_summary(
    report: WeeklyChangeReport,
    previous_summary: str = "",
) -> str:
    """使用 LLM 生成跨周变化摘要"""
    
    api_key = os.getenv("ARK_API_KEY")
    base_url = os.getenv("BASE_URL")
    model_name = os.getenv("ANALYSIS_MODEL", os.getenv("MODEL"))
    
    model = init_chat_model(
        model=model_name,
        model_provider="openai",
        api_key=api_key,
        base_url=base_url,
        temperature=0.3,
    )
    
    # 格式化变化列表
    changes_lines = []
    for c in report.changes:
        if c.significance == "significant" or c.significance == "moderate":
            changes_lines.append(
                f"- {c.metric_name}: {c.old_value} → {c.new_value} "
                f"({c.change_pct:+.1f}%) — {c.interpretation}"
            )
    
    if not changes_lines:
        return "本周各指标与上周基本持平，无显著变化。"
    
    prompt = CHANGE_SUMMARY_PROMPT.format(
        asin=report.asin,
        title=report.product_title,
        current_week=report.current_week,
        previous_week=report.previous_week,
        changes_text="\n".join(changes_lines),
        previous_summary=previous_summary or "（首次分析，无历史对比）",
    )
    
    response = model.invoke(prompt)
    return response.content if hasattr(response, 'content') else str(response)
```

## 4. 迭代分析完整流程

```python
# weekly_comparison.py 续

import httpx
from product_tools import generate_product_analysis
from product_analysis_indexer import index_product_analysis, _format_analysis_for_search


def iterate_product_analysis(asin: str, marketplace: str = "UK") -> dict:
    """
    对已有分析的产品执行迭代分析流程：
    
    1. 查询最新一周数据
    2. 查询上一次分析时的数据（从历史接口获取上一周期）
    3. 计算指标变化
    4. 生成对比分析报告（追加模式）
    5. LLM 生成变化摘要
    6. 新报告入库（追加到 Milvus，不覆盖旧报告）
    
    Returns:
        dict: 包含变化报告和新分析结果的完整信息
    """
    JAVA_BASE = os.getenv("JAVA_PRODUCT_URL", "http://java-product:8002")
    
    # 1. 获取最新数据
    current_resp = httpx.get(
        f"{JAVA_BASE}/api/v1/competitor/{asin}/detail",
        params={"marketplace": marketplace},
        timeout=10,
    )
    current_data = current_resp.json().get("data", {})
    
    # 2. 获取历史数据
    history_resp = httpx.get(
        f"{JAVA_BASE}/api/v1/competitor/{asin}/history",
        params={"marketplace": marketplace},
        timeout=10,
    )
    history_data = history_resp.json().get("data", {})
    months = history_data.get("months", [])
    
    # 至少需要两个周期的数据才能对比
    if len(months) < 2:
        # 首次分析
        analysis = generate_product_analysis(asin, marketplace)
        index_product_analysis(analysis)
        return {
            "type": "initial_analysis",
            "asin": asin,
            "analysis": analysis,
        }
    
    # 3. 本周 vs 上周对比
    current_month_data = months[-1]      # 最新一个月
    previous_month_data = months[-2]     # 上一个月
    
    product_title = current_data.get("title", "")
    
    # 4. 计算变化
    change_report = compute_weekly_changes(
        current_data=current_month_data,
        previous_data=previous_month_data,
        product_title=product_title,
        marketplace=marketplace,
    )
    
    # 5. 获取上一份分析的摘要（从 PG 或 Milvus 查）
    previous_summary = _get_previous_analysis_summary(asin)
    
    # 6. LLM 生成变化摘要
    change_summary = generate_change_summary(change_report, previous_summary)
    change_report.change_summary = change_summary
    
    # 7. 生成新的完整分析报告
    new_analysis = generate_product_analysis(asin, marketplace)
    
    # 8. 在报告中附加变化信息
    new_analysis["weekly_changes"] = {
        "current_week": change_report.current_week,
        "previous_week": change_report.previous_week,
        "changes": [
            {
                "metric": c.metric_name,
                "old": c.old_value,
                "new": c.new_value,
                "change_pct": c.change_pct,
                "significance": c.significance,
            }
            for c in change_report.changes
        ],
        "change_summary": change_summary,
    }
    new_analysis["iteration_version"] = _get_next_analysis_version(asin)
    new_analysis["previous_grade"] = previous_month_data.get("grade", "N/A")
    
    # 9. 入库（追加模式 — 新文件名，不会覆盖旧报告）
    new_analysis["week_tag"] = change_report.current_week
    index_result = index_product_analysis(new_analysis)
    
    return {
        "type": "iteration_analysis",
        "asin": asin,
        "analysis": new_analysis,
        "change_report": {
            "current_week": change_report.current_week,
            "previous_week": change_report.previous_week,
            "changes": [
                {
                    "metric": c.metric_name,
                    "old": c.old_value,
                    "new": c.new_value,
                    "change_pct": c.change_pct,
                    "significance": c.significance,
                    "interpretation": c.interpretation,
                }
                for c in change_report.changes
            ],
            "change_summary": change_summary,
        },
        "index_result": index_result,
    }


def _get_previous_analysis_summary(asin: str) -> str:
    """从 PostgreSQL 查询上一次分析的摘要"""
    from product_analysis_model import ProductAnalysisReport
    from database import SessionLocal
    
    db = SessionLocal()
    try:
        report = (
            db.query(ProductAnalysisReport)
            .filter(
                ProductAnalysisReport.asin == asin,
                ProductAnalysisReport.is_current == 1,
            )
            .order_by(ProductAnalysisReport.created_at.desc())
            .first()
        )
        return report.summary if report else ""
    finally:
        db.close()


def _get_next_analysis_version(asin: str) -> int:
    """获取该产品下一个分析版本号"""
    from product_analysis_model import ProductAnalysisReport
    from database import SessionLocal
    
    db = SessionLocal()
    try:
        count = (
            db.query(ProductAnalysisReport)
            .filter(ProductAnalysisReport.asin == asin)
            .count()
        )
        return count + 1
    finally:
        db.close()
```

```python
# 扩展 product_analysis_tasks.py

@celery_app.task(name="weekly_iterate_all_analyzed")
def weekly_iterate_all_analyzed():
    """
    每周定时任务（周日凌晨 3:00 UTC）：
    对所有已分析且本周数据已更新的产品，执行迭代分析。
    
    数据更新由 Java 后端的 ScoringService.updateWeekTags() 负责（已有）。
    本任务负责：对比 → 摘要 → 追加入库。
    """
    from database import SessionLocal
    from product_analysis_model import ProductAnalysisReport
    
    db = SessionLocal()
    try:
        # 查找所有有过分析记录且 is_current=1 的产品
        analyzed_products = (
            db.query(ProductAnalysisReport)
            .filter(ProductAnalysisReport.is_current == 1)
            .distinct(ProductAnalysisReport.asin)
            .all()
        )
        asins = list(set(p.asin for p in analyzed_products))
    finally:
        db.close()
    
    results = []
    for asin in asins:
        try:
            result = iterate_product_analysis(asin)
            results.append({"asin": asin, "status": "success", "type": result["type"]})
        except Exception as e:
            results.append({"asin": asin, "status": "error", "error": str(e)})
    
    return {
        "total": len(asins),
        "iterated": sum(1 for r in results if r["status"] == "success"),
        "errors": [r for r in results if r["status"] == "error"],
    }
```

## 6. RAG 检索：产品时间线

当用户询问一个产品的历史表现时，RAG 可以检索到所有周的分析记录。

```python
# 扩展 product_tools.py

@tool("get_product_timeline")
def get_product_timeline(asin: str) -> str:
    """
    检索某产品的完整分析时间线。
    从 Milvus 查询该 ASIN 的所有分析版本，按时间排序。
    
    Args:
        asin: 产品 ASIN
    
    Returns:
        str: 时间线概述
    """
    filter_expr = f'file_type == "product_analysis" and filename like "product_analysis_{asin}_%"'
    results = milvus_manager.query(
        filter_expr=filter_expr,
        output_fields=["text", "filename", "chunk_level"],
        limit=200,
    )
    
    if not results:
        return f"产品 {asin} 暂无分析记录。"
    
    # 按文件名（含周标签）排序
    results.sort(key=lambda x: x.get("filename", ""))
    
    # 提取每周的摘要（取 level=1 的根块作为代表）
    timeline = [f"## 产品 {asin} 分析时间线\n"]
    
    for item in results:
        if item.get("chunk_level") == 1:
            filename = item.get("filename", "")
            text = item.get("text", "")
            # 截取前 300 字符作为摘要
            snippet = text[:300] + ("..." if len(text) > 300 else "")
            timeline.append(f"### {filename}\n{snippet}\n")
    
    return "\n".join(timeline) if len(timeline) > 1 else "暂无多期分析记录。"
```

## 9. 四个阶段的完整闭环

```
Week N 数据更新
  │
  ├─ ScoringService.updateWeekTags()          [Java, 已有]
  │     └─ 标记本周 is_current=1
  │
  ├─ Celery weekly_iterate_all_analyzed       [Python, 新建]
  │     └─ 遍历已分析产品
  │         ├─ 拉取新旧数据
  │         ├─ compute_weekly_changes()       对比计算
  │         ├─ generate_change_summary()       LLM 摘要

  │
  └─ 开发者标记新产品 → 阶段一 → 阶段二 → 阶段三 → 进入迭代队列
       ↑                                                    │
       └──────────── RAG 查询已有知识指导标记决策 ←───────────┘
```
